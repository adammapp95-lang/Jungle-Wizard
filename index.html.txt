      function playTone(freq, type, duration, vol = 0.12, slideTo = null) {
        try {
          if (!audioCtx) return;
          const osc = audioCtx.createOscillator();
          const gain = audioCtx.createGain();
          osc.type = type;
          osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
          if (slideTo) {
            osc.frequency.exponentialRampToValueAtTime(slideTo, audioCtx.currentTime + duration);
          }
          gain.gain.setValueAtTime(vol, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
          osc.connect(gain);
          gain.connect(audioCtx.destination);
          osc.start();
          osc.stop(audioCtx.currentTime + duration);
        } catch (e) {}

                respawnTimer: 0
              });
            } else if (ch === 'M') {
              movingPlatforms.push({
                x,
                y,
                w: TILE * 2.5,
                h: 18,
                startX: x - 48,
                endX: x + 130,
                startY: y,
                endY: y,
                vx: 1.5,
                vy: 0
              });
            } else if (ch === 'V') {
              movingPlatforms.push({
                x,
                y,
                w: TILE * 2.2,
                h: 18,
                startX: x,
                endX: x,
                startY: y - 92,
                endY: y + 72,
                vx: 0,
                vy: -1.35
              });
            } else if (ch === 'G') {
              gems.push({
                x: x + 8,
                y: y + 8,
                w: 16,
                h: 16,
                color: gemColors[(r + c) % gemColors.length],
                bob: Math.random() * Math.PI * 2
              });
            } else if (ch === 'H') {
              flowers.push({
                x: x + 6,
                y: y + 6,
                w: 20,
                h: 20,
                bob: Math.random() * Math.PI * 2
              });
            } else if (ch === 'E') {
              enemies.push({
                x,
                y,
                w: 34,
                h: 22,
                startX: x - 95,
                endX: x + 95,
                vx: 1.7,
                shootTimer: 45 + Math.floor(Math.random() * 40),
                flap: 0
              });
            } else if (ch === 'S') {
              spawnPoint = { x, y: y - 4 };
            } else if (ch === 'P') {
              portal = { x: x - 4, y: y - 24, w: 36, h: 56 };
            }
          }
        }

        // Reset Player
        player.x = spawnPoint.x;
        player.y = spawnPoint.y;
        player.vx = 0;
        player.vy = 0;
        player.grounded = false;
        player.jumpsLeft = 2;
        player.facing = 1;
        player.maxHp = lvl.maxHp;
        player.hp = lvl.startHp;
        player.invuln = 0;
        player.shieldTimer = 0;
        player.shieldCooldown = 0;
        cameraX = Math.max(0, player.x - VIEW_W * 0.3);

        updateHud();
      }

      function respawnAtLevelStart() {
        player.x = spawnPoint.x;
        player.y = spawnPoint.y;
        player.vx = 0;
        player.vy = 0;
        player.grounded = false;
        player.jumpsLeft = 2;
        player.invuln = 60;
        cameraX = Math.max(0, player.x - VIEW_W * 0.3);
        // Restore any collapsed crumbling blocks so level remains traversable
        for (const cb of crumblingBlocks) {
          cb.state = 'solid';
          cb.timer = 0;
          cb.respawnTimer = 0;
        }
        spawnParticles(player.x + player.w / 2, player.y + player.h / 2, '#74c69d', 16, 3.5);
      }

      function damagePlayer(knockDir = 0, fromPit = false) {
        if (!fromPit && (player.invuln > 0 || player.shieldTimer > 0)) return;

        player.hp = Math.max(0, player.hp - 1);
        sfx.hurt();
        updateHud();

        if (player.hp <= 0) {
          state = 'DYING';
          deathTimer = 55;
          spawnParticles(player.x + player.w / 2, player.y + player.h / 2, '#ff5400', 26, 4.5);
          return;
        }

        if (fromPit) {
          // Falling off the map deducts 1 HP and places the wizard back at the start of the current level
          respawnAtLevelStart();
        } else {
          player.invuln = 65;
          player.vy = -5.5;
          player.vx = knockDir * 4.5;
          spawnParticles(player.x + player.w / 2, player.y + player.h / 2, '#ff2a55', 14, 3);
        }
      }

      function triggerJump() {
        if (state !== 'PLAYING') return;
        if (player.grounded || player.ledgeGrab) {
          player.vy = player.jumpForce;
          player.grounded = false;
          player.ledgeGrab = false;
          player.jumpsLeft = 1;
          sfx.jump();
          spawnParticles(player.x + player.w / 2, player.y + player.h, '#95d5b2', 7, 2);
        } else if (player.jumpsLeft > 0) {
          player.vy = player.jumpForce * 0.92;
          player.jumpsLeft--;
          sfx.doubleJump();
          spawnParticles(player.x + player.w / 2, player.y + player.h, '#ffd166', 10, 2.5);
        }
      }

      function triggerShield() {
        if (state !== 'PLAYING') return;
        if (!LEVELS[currentLevelIdx].hasShield) return;
        if (player.shieldCooldown > 0 || player.shieldTimer > 0) return;
        player.shieldTimer = 150; // 2.5s active fiery force field
        player.shieldCooldown = 270; // 4.5s cooldown
        sfx.shield();
        spawnParticles(player.x + player.w / 2, player.y + player.h / 2, '#ff7b00', 18, 4);
        updateHud();
      }

      function rectOverlap(a, b) {
        return (
          a.x < b.x + b.w &&
          a.x + a.w > b.x &&
          a.y < b.y + b.h &&
          a.y + a.h > b.y
        );
      }

      function showOverlay(overlayEl) {
        [startScreen, pauseScreen, levelClearScreen, gameoverScreen, victoryScreen].forEach(el => {
          el.classList.add('hidden');
        });
        if (overlayEl) overlayEl.classList.remove('hidden');
      }

      function startLevel(idx, keepScore = false) {
        currentLevelIdx = idx;
        if (!keepScore) {
          score = 0;
        }
        levelStartScore = score;
        saveProgress(currentLevelIdx, levelStartScore);
        buildLevel(currentLevelIdx);
        showOverlay(null);
        hud.style.display = 'flex';
        touchControls.style.display = 'flex';
        state = 'PLAYING';
      }

      function update() {
        frameCount++;

        // Update particles even during dying sequence
        for (let i = particles.length - 1; i >= 0; i--) {
          const p = particles[i];
          p.x += p.vx;
          p.y += p.vy;
          p.vy += 0.12;
          p.life--;
          if (p.life <= 0) particles.splice(i, 1);
        }

        if (state === 'DYING') {
          deathTimer--;
          if (deathTimer <= 0) {
            state = 'GAMEOVER';
            hud.style.display = 'none';
            touchControls.style.display = 'none';
            document.getElementById('gameover-text').textContent =
              `Fallen on ${LEVELS[currentLevelIdx].name}. Try again!`;
            retryBtn.textContent = `RETRY LEVEL ${currentLevelIdx + 1}`;
            showOverlay(gameoverScreen);
          }
          return;
        }

        if (state !== 'PLAYING') return;

        // Timers
        if (player.invuln > 0) player.invuln--;
        if (player.shieldTimer > 0) player.shieldTimer--;
        if (player.shieldCooldown > 0) {
          player.shieldCooldown--;
          if (player.shieldCooldown === 0) updateHud();
        }

        // Moving Platforms Update
        for (const mp of movingPlatforms) {
          const prevX = mp.x;
          const prevY = mp.y;
          mp.x += mp.vx;
          mp.y += mp.vy;
          if (mp.vx !== 0 && (mp.x <= mp.startX || mp.x >= mp.endX)) mp.vx *= -1;
          if (mp.vy !== 0 && (mp.y <= mp.startY || mp.y >= mp.endY)) mp.vy *= -1;
          mp.dx = mp.x - prevX;
          mp.dy = mp.y - prevY;
        }

        // Crumbling Bridges Update
        for (const cb of crumblingBlocks) {
          if (cb.state === 'shaking') {
            cb.timer--;
            if (cb.timer <= 0) {
              cb.state = 'fallen';
              cb.respawnTimer = 210; // Respawns after 3.5 seconds
              sfx.crumble();
              spawnParticles(cb.x + cb.w / 2, cb.y + 8, '#8d99ae', 10, 2.5);
            }
          } else if (cb.state === 'fallen') {
            cb.respawnTimer--;
            if (cb.respawnTimer <= 0) {
              cb.state = 'solid';
            }
          }
        }

        // Horizontal Movement
        if (keys.left) {
          player.vx = -player.speed;
          player.facing = -1;
        } else if (keys.right) {
          player.vx = player.speed;
          player.facing = 1;
        } else {
          player.vx *= 0.72;
          if (Math.abs(player.vx) < 0.15) player.vx = 0;
        }

        player.x += player.vx;
        if (player.x < 0) player.x = 0;
        if (player.x + player.w > worldWidth) player.x = worldWidth - player.w;

        // Horizontal Solid Collisions
        player.ledgeGrab = false;
        for (const s of solids) {
          if (s.type === 'ground' && rectOverlap(player, s)) {
            if (player.vx > 0) {
              player.x = s.x - player.w;
              // Ledge grab check
              if (!player.grounded && player.vy > 0 && Math.abs(player.y - s.y) < 14) {
                player.vy = 0;
                player.ledgeGrab = true;
                player.jumpsLeft = 1;
              }
            } else if (player.vx < 0) {
              player.x = s.x + s.w;
              if (!player.grounded && player.vy > 0 && Math.abs(player.y - s.y) < 14) {
                player.vy = 0;
                player.ledgeGrab = true;
                player.jumpsLeft = 1;
              }
            }
          }
        }

        // Gravity & Vertical Movement
        const prevBottom = player.y + player.h;
        player.vy = Math.min(player.vy + 0.44, 11.5);
        player.y += player.vy;
        player.grounded = false;

        // Vertical Solid Collisions
        for (const s of solids) {
          if (s.type === 'ground') {
            if (rectOverlap(player, s)) {
              if (player.vy >= 0 && prevBottom <= s.y + 10) {
                player.y = s.y - player.h;
                player.vy = 0;
                player.grounded = true;
                player.jumpsLeft = 2;
              } else if (player.vy < 0 && player.y >= s.y + s.h - 10) {
                player.y = s.y + s.h;
                player.vy = 0;
              }
            }
          } else if (s.type === 'ledge') {
            if (player.vy >= 0 && prevBottom <= s.y + 8 && rectOverlap(player, s)) {
              player.y = s.y - player.h;
              player.vy = 0;
              player.grounded = true;
              player.jumpsLeft = 2;
            }
          }
        }

        // Crumbling Bridges Collision
        for (const cb of crumblingBlocks) {
          if (cb.state !== 'fallen' && player.vy >= 0 && prevBottom <= cb.y + 10 && rectOverlap(player, cb)) {
            player.y = cb.y - player.h;
            player.vy = 0;
            player.grounded = true;
            player.jumpsLeft = 2;
            if (cb.state === 'solid') {
              cb.state = 'shaking';
              cb.timer = 26; // Collapses quickly!
            }
          }
        }

        // Moving Platforms Collision (Carry Player)
        for (const mp of movingPlatforms) {
          if (player.vy >= 0 && prevBottom <= mp.y + 12 && rectOverlap(player, mp)) {
            player.y = mp.y - player.h;
            player.vy = 0;
            player.grounded = true;
            player.jumpsLeft = 2;
            player.x += mp.dx;
            player.y += mp.dy;
          }
        }

        // Pit Fall Check -> Deduct 1 HP and send back to start of current level
        if (player.y > worldHeight + 24) {
          damagePlayer(0, true);
          return;
        }

        // Grounded Neon-Magenta Crystal Spikes Check
        for (const sp of spikes) {
          if (rectOverlap(player, sp)) {
            const dir = player.x + player.w / 2 < sp.x + sp.w / 2 ? -1 : 1;
            damagePlayer(dir, false);
            break;
          }
        }

        // Gems Collection
        for (let i = gems.length - 1; i >= 0; i--) {
          const g = gems[i];
          g.bob += 0.08;
          if (rectOverlap(player, g)) {
            gems.splice(i, 1);
            score += 10;
            sfx.gem();
            spawnParticles(g.x + 8, g.y + 8, g.color, 10, 2.5);
            updateHud();
          }
        }

        // Healing Flowers Collection
        for (let i = flowers.length - 1; i >= 0; i--) {
          const f = flowers[i];
          f.bob += 0.07;
          if (rectOverlap(player, f)) {
            flowers.splice(i, 1);
            if (player.hp < player.maxHp) {
              player.hp++;
            } else {
              score += 25;
            }
            sfx.heal();
            spawnParticles(f.x + 10, f.y + 10, '#ff8fab', 12, 2.8);
            updateHud();
          }
        }

        // Pterodactyl Enemies & Forward-Facing Toxic Venom Projectiles
        for (let i = enemies.length - 1; i >= 0; i--) {
          const e = enemies[i];
          e.flap += 0.22;
          e.x += e.vx;
          if (e.x <= e.startX) {
            e.x = e.startX;
            e.vx = Math.abs(e.vx);
            e.shootTimer = Math.min(e.shootTimer, 18); // Quick reaction shot when turning toward player
          } else if (e.x >= e.endX) {
            e.x = e.endX;
            e.vx = -Math.abs(e.vx);
            e.shootTimer = Math.min(e.shootTimer, 18);
          }

          // Fiery Force Field destroys nearby pterodactyls
          if (player.shieldTimer > 0) {
            const shieldBox = { x: player.x - 16, y: player.y - 16, w: player.w + 32, h: player.h + 32 };
            if (rectOverlap(shieldBox, e)) {
              enemies.splice(i, 1);
              score += 30;
              sfx.shield();
              spawnParticles(e.x + e.w / 2, e.y + e.h / 2, '#ff7b00', 18, 4);
              updateHud();
              continue;
            }
          }

          if (rectOverlap(player, e)) {
            const dir = player.x + player.w / 2 < e.x + e.w / 2 ? -1 : 1;
            damagePlayer(dir, false);
          }

          // Shoot toxic-green venom ONLY when flying/facing toward the wizard
          const facingPlayer = (e.vx > 0 && player.x > e.x) || (e.vx < 0 && player.x < e.x);
          const distToPlayer = Math.hypot(player.x - e.x, player.y - e.y);

          e.shootTimer--;
          if (e.shootTimer <= 0 && facingPlayer && distToPlayer < 430) {
            e.shootTimer = 68;
            const angle = Math.atan2(player.y + 10 - (e.y + 12), player.x + 10 - (e.x + e.w / 2));
            const spd = 4.3;
            projectiles.push({
              x: e.x + (e.vx > 0 ? e.w : 0),
              y: e.y + 12,
              w: 10,
              h: 10,
              vx: Math.cos(angle) * spd,
              vy: Math.sin(angle) * spd
            });
            sfx.venom();
          }
        }

        // Update Toxic Venom Projectiles
        for (let i = projectiles.length - 1; i >= 0; i--) {
          const p = projectiles[i];
          p.x += p.vx;
          p.y += p.vy;
          if (frameCount % 3 === 0) {
            spawnParticles(p.x + 5, p.y + 5, '#39ff14', 1, 0.8);
          }

          if (player.shieldTimer > 0) {
            const shieldBox = { x: player.x - 16, y: player.y - 16, w: player.w + 32, h: player.h + 32 };
            if (rectOverlap(shieldBox, p)) {
              projectiles.splice(i, 1);
              spawnParticles(p.x + 5, p.y + 5, '#ff9e00', 8, 2.5);
              continue;
            }
          }

          if (rectOverlap(player, p)) {
            projectiles.splice(i, 1);
            damagePlayer(p.vx > 0 ? 1 : -1, false);
            continue;
          }

          if (p.x < cameraX - 100 || p.x > cameraX + VIEW_W + 100 || p.y > worldHeight + 40) {
            projectiles.splice(i, 1);
          }
        }

        // Portal Check
        if (portal && rectOverlap(player, portal)) {
          sfx.portal();
          spawnParticles(portal.x + portal.w / 2, portal.y + portal.h / 2, '#00f5d4', 28, 4.5);
          if (currentLevelIdx + 1 < LEVELS.length) {
            state = 'LEVEL_CLEAR';
            saveProgress(currentLevelIdx + 1, score);
            hud.style.display = 'none';
            touchControls.style.display = 'none';
            document.getElementById('level-clear-text').textContent =
              `Completed ${LEVELS[currentLevelIdx].name}! Next: ${LEVELS[currentLevelIdx + 1].name}`;
            showOverlay(levelClearScreen);
          } else {
            state = 'VICTORY';
            clearProgress();
            hud.style.display = 'none';
            touchControls.style.display = 'none';
            document.getElementById('victory-text').textContent =
              `All 4 Jungle Trials Conquered! Final Gem Score: ${score}`;
            showOverlay(victoryScreen);
          }
          return;
        }

        // Smooth Camera Follow
        const targetCam = player.x - VIEW_W * 0.36;
        cameraX += (targetCam - cameraX) * 0.12;
        cameraX = Math.max(0, Math.min(cameraX, worldWidth - VIEW_W));
      }

      function render() {
        const lvl = LEVELS[currentLevelIdx];

        // Jungle Sky Gradient
        const grad = ctx.createLinearGradient(0, 0, 0, VIEW_H);
        grad.addColorStop(0, lvl.bgTop);
        grad.addColorStop(1, lvl.bgBottom);
        ctx.fillStyle = grad;
        ctx.fillRect(0, 0, VIEW_W, VIEW_H);

        // Parallax Jungle Silhouettes & Vines
        ctx.fillStyle = 'rgba(8, 22, 13, 0.45)';
        for (let i = 0; i < 14; i++) {
          const px = ((i * 180 - cameraX * 0.25) % (VIEW_W + 240)) - 120;
          ctx.fillRect(px, 40, 26, VIEW_H);
          ctx.beginPath();
          ctx.arc(px + 13, 70, 56, 0, Math.PI * 2);
          ctx.fill();
        }

        ctx.save();
        ctx.translate(-Math.floor(cameraX), 0);

        // Render Portal
        if (portal) {
          const pulse = Math.sin(frameCount * 0.1) * 4;
          ctx.fillStyle = '#3a0ca3';
          ctx.beginPath();
          ctx.ellipse(portal.x + portal.w / 2, portal.y + portal.h / 2, 18 + pulse, 28 + pulse, 0, 0, Math.PI * 2);
          ctx.fill();

          ctx.fillStyle = '#00f5d4';
          ctx.beginPath();
          ctx.ellipse(portal.x + portal.w / 2, portal.y + portal.h / 2, 10 + pulse * 0.5, 20 + pulse * 0.5, 0, 0, Math.PI * 2);
          ctx.fill();
        }

        // Render Solids (Jungle Earth & Temple Stone Ledges)
        for (const s of solids) {
          if (s.x + s.w < cameraX - 40 || s.x > cameraX + VIEW_W + 40) continue;
          if (s.type === 'ground') {
            ctx.fillStyle = '#2d2a26';
            ctx.fillRect(s.x, s.y, s.w, s.h);
            ctx.fillStyle = '#3f3b36';
            ctx.fillRect(s.x + 2, s.y + 6, s.w - 4, s.h - 8);
            // Lush Mossy Top
            ctx.fillStyle = '#2d6a4f';
            ctx.fillRect(s.x, s.y, s.w, 7);
            ctx.fillStyle = '#52b788';
            ctx.fillRect(s.x, s.y, s.w, 3);
          } else {
            // Carved Stone Ledge
            ctx.fillStyle = '#4a4e4d';
            ctx.fillRect(s.x, s.y, s.w, s.h);
            ctx.fillStyle = '#40916c';
            ctx.fillRect(s.x, s.y, s.w, 4);
          }
        }

        // Render Authentic Mossy Temple-Stone Moving Platforms
        for (const mp of movingPlatforms) {
          ctx.fillStyle = '#3d403d';
          ctx.fillRect(mp.x, mp.y, mp.w, mp.h);
          // Stone brick mortar lines
          ctx.fillStyle = '#242624';
          ctx.fillRect(mp.x + mp.w * 0.33, mp.y + 4, 2, mp.h - 4);
          ctx.fillRect(mp.x + mp.w * 0.66, mp.y + 4, 2, mp.h - 4);
          // Lush jungle moss top
          ctx.fillStyle = '#2d6a4f';
          ctx.fillRect(mp.x, mp.y, mp.w, 6);
          ctx.fillStyle = '#74c69d';
          ctx.fillRect(mp.x + 2, mp.y, mp.w - 4, 2);
          // Small hanging jungle vines underneath
          ctx.fillStyle = '#40916c';
          ctx.fillRect(mp.x + 10, mp.y + mp.h, 3, 7);
          ctx.fillRect(mp.x + mp.w - 14, mp.y + mp.h, 3, 5);
        }

        // Render Crumbling Stone Bridges
        for (const cb of crumblingBlocks) {
          if (cb.state === 'fallen') continue;
          const shakeX = cb.state === 'shaking' ? (Math.random() - 0.5) * 4 : 0;
          ctx.fillStyle = cb.state === 'shaking' ? '#9e2a2b' : '#5c5852';
          ctx.fillRect(cb.x + shakeX, cb.y, cb.w - 2, cb.h);
          // Cracks
          ctx.strokeStyle = '#1b1a18';
          ctx.lineWidth = 2;
          ctx.beginPath();
          ctx.moveTo(cb.x + shakeX + 6, cb.y + 2);
          ctx.lineTo(cb.x + shakeX + 14, cb.y + 10);
          ctx.lineTo(cb.x + shakeX + 10, cb.y + 16);
          ctx.stroke();
        }

        // Render Grounded Pulsing Neon-Magenta Crystal Spikes
        const spikeGlow = 0.75 + Math.sin(frameCount * 0.16) * 0.25;
        for (const sp of spikes) {
          if (sp.x + sp.w < cameraX - 40 || sp.x > cameraX + VIEW_W + 40) continue;
          ctx.fillStyle = `rgba(255, 0, 127, ${spikeGlow})`;
          const count = 3;
          const step = sp.w / count;
          for (let i = 0; i < count; i++) {
            const sx = sp.x + i * step;
            ctx.beginPath();
            ctx.moveTo(sx, sp.y + sp.h);
            ctx.lineTo(sx + step * 0.5, sp.y);
            ctx.lineTo(sx + step, sp.y + sp.h);
            ctx.closePath();
            ctx.fill();
          }
        }

        // Render Gems
        for (const g of gems) {
          const gy = g.y + Math.sin(g.bob) * 3;
          ctx.fillStyle = g.color;
          ctx.beginPath();
          ctx.moveTo(g.x + 8, gy);
          ctx.lineTo(g.x + 16, gy + 8);
          ctx.lineTo(g.x + 8, gy + 16);
          ctx.lineTo(g.x, gy + 8);
          ctx.closePath();
          ctx.fill();
        }

        // Render Healing Lotus Flowers
        for (const f of flowers) {
          const fy = f.y + Math.sin(f.bob) * 2.5;
          ctx.fillStyle = '#2d6a4f';
          ctx.fillRect(f.x + 8, fy + 10, 4, 10);
          ctx.fillStyle = '#ff7096';
          ctx.beginPath();
          ctx.arc(f.x + 10, fy + 8, 7, 0, Math.PI * 2);
          ctx.fill();
          ctx.fillStyle = '#ffd166';
          ctx.beginPath();
          ctx.arc(f.x + 10, fy + 8, 3, 0, Math.PI * 2);
          ctx.fill();
        }

        // Render Pterodactyls
        for (const e of enemies) {
          const wingY = Math.sin(e.flap) * 7;
          ctx.fillStyle = '#7b2cbf';
          // Body
          ctx.fillRect(e.x + 6, e.y + 8, 22, 10);
          // Beak facing movement direction
          ctx.fillStyle = '#ff9e00';
          ctx.beginPath();
          if (e.vx >= 0) {
            ctx.moveTo(e.x + 28, e.y + 9);
            ctx.lineTo(e.x + 38, e.y + 13);
            ctx.lineTo(e.x + 28, e.y + 17);
          } else {
            ctx.moveTo(e.x + 6, e.y + 9);
            ctx.lineTo(e.x - 4, e.y + 13);
            ctx.lineTo(e.x + 6, e.y + 17);
          }
          ctx.closePath();
          ctx.fill();
          // Wings
          ctx.fillStyle = '#9d4edd';
          ctx.beginPath();
          ctx.moveTo(e.x + 12, e.y + 10);
          ctx.lineTo(e.x + 18, e.y - 4 + wingY);
          ctx.lineTo(e.x + 24, e.y + 10);
          ctx.closePath();
          ctx.fill();
        }

        // Render Toxic-Green Venom Projectiles
        for (const p of projectiles) {
          ctx.fillStyle = '#39ff14';
          ctx.beginPath();
          ctx.arc(p.x + 5, p.y + 5, 5.5, 0, Math.PI * 2);
          ctx.fill();
          ctx.fillStyle = '#d8f3dc';
          ctx.beginPath();
          ctx.arc(p.x + 4, p.y + 4, 2, 0, Math.PI * 2);
          ctx.fill();
        }

        // Render Wizard Player (or fiery skeleton if dying)
        if (state === 'DYING') {
          ctx.fillStyle = '#f8f9fa';
          ctx.fillRect(player.x + 6, player.y + 4, 10, 10); // Skull
          ctx.fillRect(player.x + 9, player.y + 14, 4, 12); // Spine
          ctx.fillRect(player.x + 4, player.y + 16, 14, 2); // Ribs
        } else {
          const blink = player.invuln > 0 && Math.floor(player.invuln / 4) % 2 === 0;
          if (!blink) {
            // Red & Black Wizard Robe
            ctx.fillStyle = '#1b1b1e';
            ctx.fillRect(player.x + 3, player.y + 11, 16, 19);
            ctx.fillStyle = '#d90429';
            ctx.fillRect(player.x + 5, player.y + 12, 12, 17);

            // Wizard Head & Glowing Eyes
            ctx.fillStyle = '#ffccd5';
            ctx.fillRect(player.x + 6, player.y + 5, 10, 7);
            ctx.fillStyle = '#00f5d4';
            const eyeX = player.facing === 1 ? player.x + 12 : player.x + 7;
            ctx.fillRect(eyeX, player.y + 7, 3, 2);

            // Pointed Red Wizard Hat
            ctx.fillStyle = '#ef233c';
            ctx.beginPath();
            ctx.moveTo(player.x + 1, player.y + 6);
            ctx.lineTo(player.x + 11, player.y - 8);
            ctx.lineTo(player.x + 21, player.y + 6);
            ctx.closePath();
            ctx.fill();

            // Arcane Staff
            const staffX = player.facing === 1 ? player.x + 20 : player.x;
            ctx.fillStyle = '#8d5b4c';
            ctx.fillRect(staffX, player.y + 8, 2, 22);
            ctx.fillStyle = '#ffd166';
            ctx.fillRect(staffX - 1, player.y + 5, 4, 4);
          }

          // Active Fiery Force Field Aura
          if (player.shieldTimer > 0) {
            ctx.strokeStyle = frameCount % 4 < 2 ? '#ff7b00' : '#ffd166';
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.arc(player.x + player.w / 2, player.y + player.h / 2, 24, 0, Math.PI * 2);
            ctx.stroke();
          }
        }

        // Render Particles
        for (const p of particles) {
          ctx.globalAlpha = Math.max(0, p.life / p.maxLife);
          ctx.fillStyle = p.color;
          ctx.fillRect(p.x, p.y, p.size, p.size);
        }
        ctx.globalAlpha = 1;

        ctx.restore();
      }

      function loop() {
        update();
        render();
        requestAnimationFrame(loop);
      }

      // Unified Tap/Click Helper
      function bindButton(el, handler) {
        let lastTap = 0;
        const wrapped = (e) => {
          if (e) {
            e.preventDefault();
            e.stopPropagation();
          }
          const now = Date.now();
          if (now - lastTap < 250) return;
          lastTap = now;
          initAudio();
          handler();
        };
        el.addEventListener('touchstart', wrapped, { passive: false });
        el.addEventListener('pointerdown', wrapped);
        el.addEventListener('click', wrapped);
      }

      // Bind Menu Buttons
      bindButton(startBtn, () => {
        clearProgress();
        startLevel(0, false);
      });

      bindButton(continueBtn, () => {
        const saved = loadProgress();
        if (saved) {
          score = saved.score || 0;
          startLevel(saved.level, true);
        } else {
          startLevel(0, false);
        }
      });

      bindButton(pauseBtn, () => {
        if (state === 'PLAYING') {
          state = 'PAUSED';
          document.getElementById('pause-level-label').textContent = LEVELS[currentLevelIdx].name;
          showOverlay(pauseScreen);
        }
      });

      bindButton(resumeBtn, () => {
        if (state === 'PAUSED') {
          showOverlay(null);
          state = 'PLAYING';
        }
      });

      bindButton(restartLevelBtn, () => {
        score = levelStartScore;
        startLevel(currentLevelIdx, true);
      });

      bindButton(quitTitleBtn, () => {
        saveProgress(currentLevelIdx, levelStartScore);
        state = 'START';
        hud.style.display = 'none';
        touchControls.style.display = 'none';
        refreshContinueButton();
        showOverlay(startScreen);
      });

      bindButton(nextLevelBtn, () => {
        startLevel(currentLevelIdx + 1, true);
      });

      bindButton(retryBtn, () => {
        score = levelStartScore;
        startLevel(currentLevelIdx, true);
      });

      bindButton(gameoverTitleBtn, () => {
        state = 'START';
        hud.style.display = 'none';
        touchControls.style.display = 'none';
        refreshContinueButton();
        showOverlay(startScreen);
      });

      bindButton(playAgainBtn, () => {
        clearProgress();
        startLevel(0, false);
      });

      // Touch Controls Binding
      function bindHoldButton(el, onDown, onUp) {
        const press = (e) => {
          e.preventDefault();
          initAudio();
          el.classList.add('pressed');
          onDown();
        };
        const release = (e) => {
          e.preventDefault();
          el.classList.remove('pressed');
          if (onUp) onUp();
        };
        el.addEventListener('touchstart', press, { passive: false });
        el.addEventListener('touchend', release, { passive: false });
        el.addEventListener('touchcancel', release, { passive: false });
        el.addEventListener('mousedown', press);
        el.addEventListener('mouseup', release);
        el.addEventListener('mouseleave', release);
      }

      bindHoldButton(
        document.getElementById('btn-left'),
        () => { keys.left = true; },
        () => { keys.left = false; }
      );

      bindHoldButton(
        document.getElementById('btn-right'),
        () => { keys.right = true; },
        () => { keys.right = false; }
      );

      bindHoldButton(
        document.getElementById('btn-jump'),
        () => { triggerJump(); },
        null
      );

      bindHoldButton(
        btnShield,
        () => { triggerShield(); },
        null
      );

      // Keyboard Controls
      window.addEventListener('keydown', (e) => {
        if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = true;
        if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = true;
        if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') {
          if (!keys.jump) triggerJump();
          keys.jump = true;
        }
        if (e.code === 'ShiftLeft' || e.code === 'ShiftRight' || e.code === 'KeyF') {
          triggerShield();
        }
        if ((e.code === 'Escape' || e.code === 'KeyP') && state === 'PLAYING') {
          state = 'PAUSED';
          document.getElementById('pause-level-label').textContent = LEVELS[currentLevelIdx].name;
          showOverlay(pauseScreen);
        }
      });

      window.addEventListener('keyup', (e) => {
        if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = false;
        if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = false;
        if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') keys.jump = false;
      });

      // Boot
      refreshContinueButton();
      buildLevel(0);
      requestAnimationFrame(loop);
    })();
  </script>
